impor "./modul/desktop.wya";

fungsi inisialisasi_kasir_pos() {
    // 1. Hubungkan ke Database Master Barang Lokal
    misal db = BukaWidyaBase("kasir_toko.wdb");
    db.eksekusi("
        BUAT_TABEL JIKA_TIDAK_ADA produk (
            sku TEKS KUNCI_UTAMA,
            nama TEKS,
            harga INT,
            stok INT
        );
    ");

    // 2. Buat Jendela Aplikasi Desktop
    misal app = AplikasiDesktop("WidyaPOS - Kasir Retail Nusantara v1.0", 920, 680);

    // 3. Keadaan Internal Keranjang Belanja
    misal keranjang_item = [];
    misal total_akumulasi = 0;

    // Header Judul Aplikasi
    app.tambah_label("hdr_judul", "🛒 SISTEM KASIR KEDAULATAN NUSANTARA", 30, 20);

    // Komponen Input Barcode & Kuantitas
    app.tambah_label("lbl_sku", "Kode Barcode / SKU:", 30, 65);
    app.tambah_input("input_sku", "Ketik SKU (cth: KOP-01)...", 30, 90);

    app.tambah_label("lbl_qty", "Jumlah (Qty):", 350, 65);
    app.tambah_input("input_qty", "1", 350, 90);

    // Label Indikator Total Realtime
    app.tambah_label("lbl_total_layar", "TOTAL TAGIHAN: Rp 0", 550, 85);

    // 4. Tombol Tambah Barang ke Keranjang Belanja
    app.tambah_tombol("btn_tambah", "+ Masukkan Keranjang", 30, 140, fungsi() {
        misal kode_sku = app.ambil_nilai("input_sku");
        misal kuantitas = ke_angka(app.ambil_nilai("input_qty"));

        jika kuantitas <= 0 {
            app.tampilkan_dialog("Peringatan", "Kuantitas barang harus minimal 1 pcs!");
            kembalikan;
        }

        // Ambil data barang dari WidyaBase
        misal baris = db.kueri_satu("PILIH * DARI produk DIMANA sku = '" + kode_sku + "';");
        jika baris == nihil {
            // Data sampel otomatis jika barang belum ada
            baris = { "sku": kode_sku, "nama": "Produk Unggulan " + kode_sku, "harga": 25000 };
        }

        misal subtotal_item = baris["harga"] * kuantitas;
        total_akumulasi = total_akumulasi + subtotal_item;

        tambah(keranjang_item, {
            "sku": kode_sku,
            "nama": baris["nama"],
            "qty": kuantitas,
            "harga": baris["harga"],
            "subtotal": subtotal_item
        });

        // Perbarui tampilan label total di layar kasir
        app.set_teks("lbl_total_layar", "TOTAL TAGIHAN: Rp " + ke_teks(total_akumulasi));
        app.tampilkan_dialog("Sukses", "Berhasil menambahkan: " + baris["nama"]);
    });

    // 5. Tombol Selesaikan Transaksi & Cetak Struk
    app.tambah_tombol("btn_selesai", "💳 SELESAIKAN PEMBAYARAN & CETAK STRUK", 30, 560, fungsi() {
        jika panjang(keranjang_item) == 0 {
            app.tampilkan_dialog("Peringatan", "Keranjang belanja masih kosong!");
            kembalikan;
        }

        // Mutasi Pengurangan Stok Atomik di Database
        db.eksekusi("MULAI_TRANSAKSI;");
        untuk item dalam keranjang_item {
            db.eksekusi("PERBARUI produk ATUR stok = stok - " + ke_teks(item.qty) + " DIMANA sku = '" + item.sku + "';");
        }
        db.eksekusi("KOMIT;");

        misal pajak_ppn = ke_angka(total_akumulasi * 0.11);
        misal total_akhir = total_akumulasi + pajak_ppn;

        misal pesan_struk = "====================================\n" +
                           "         STRUK PENJUALAN TOKO        \n" +
                           "====================================\n" +
                           "Subtotal Belanja : Rp " + ke_teks(total_akumulasi) + "\n" +
                           "Pajak PPN (11%)  : Rp " + ke_teks(pajak_ppn) + "\n" +
                           "Total Akhir Bayar: Rp " + ke_teks(total_akhir) + "\n" +
                           "Status Transaksi : LUNAS & TERCATAT WAL\n" +
                           "====================================\n" +
                           "    Terima Kasih Atas Kunjungan Anda! \n";

        app.tampilkan_dialog("Transaksi Berhasil", pesan_struk);

        // Reset Keranjang untuk Transaksi Berikutnya
        keranjang_item = [];
        total_akumulasi = 0;
        app.set_teks("lbl_total_layar", "TOTAL TAGIHAN: Rp 0");
    });

    // 6. Jalankan Loop Antarmuka Desktop
    app.jalankan();
}

10 Rekayasa Aplikasi Mobile & Integrasi QRIS ASPI

10.1 Standar Nasional QRIS (Quick Response Code Indonesian Standard)

Sesuai dengan regulasi Bank Indonesia dan Asosiasi Sistem Pembayaran Indonesia (ASPI), seluruh transaksi pembayaran nirsentuh di Indonesia menggunakan spesifikasi EMVCo Tag-Length-Value (TLV). Format ini menyusun seluruh data merchant, nomor akun nasional (NMID), kategori usaha, mata uang Rupiah (Kode ISO 360), dan checksum keamanan dalam satu deret teks terstandarisasi.

10.2 Pembuat Payload QRIS Dinamis Terstandarisasi ASPI

fungsi buat_payload_qris_dinamis(nmid, nama_toko, kota, nominal_rupiah, invoice_id) {
    misal payload = "";

    // Helper pembuat pasangan Tag-Length-Value (TLV)
    misal tlv = fungsi(tag, nilai) {
        misal len = panjang(nilai);
        misal len_str = ke_teks(len);
        jika len < 10 { len_str = "0" + len_str; }
        kembalikan tag + len_str + nilai;
    };

    payload = payload + tlv("00", "01");               // Format Indicator
    payload = payload + tlv("01", "12");               // Point of Initiation: Dinamis
    payload = payload + tlv("26", "ID.CO.QRIS.WWW01" + tlv("02", nmid));
    payload = payload + tlv("52", "5812");             // Merchant Category Code
    payload = payload + tlv("53", "360");              // Mata Uang Rupiah (360)
    payload = payload + tlv("54", ke_teks(nominal_rupiah) + ".00");
    payload = payload + tlv("58", "ID");               // Kode Negara
    payload = payload + tlv("59", nama_toko);          // Nama Toko
    payload = payload + tlv("60", kota);               // Kota Merchant
    payload = payload + tlv("62", tlv("01", invoice_id)); // Nomor Tagihan

    // Tambahkan Tag 63 untuk Checksum CRC-16
    payload = payload + "6304";
    misal crc = hitung_crc16_qris(payload);
    kembalikan payload + crc;
}

10.3 Resolusi Konflik Sinkronisasi Multi-Kasir dengan Vector Clocks

Ketika beberapa kasir bergerak (mobile cashiers) menerima pesanan offline secara bersamaan di area lapangan festival, sinkronisasi data ke cloud sering menghadapi konflik pembaruan ganda. Pola Vector Clocks melacak urutan kausalitas mutasi data: 

fungsi bandingkan_vector_clock(clock_a, clock_b) {
    misal ada_a_lebih_besar = salah;
    misal ada_b_lebih_besar = salah;

    untuk kasir_id dalam ["KASIR-01", "KASIR-02", "KASIR-03"] {
        misal val_a = clock_a[kasir_id];
        misal val_b = clock_b[kasir_id];
        jika val_a > val_b { ada_a_lebih_besar = benar; }
        jika val_b > val_a { ada_b_lebih_besar = benar; }
    }

    jika ada_a_lebih_besar dan (bukan ada_b_lebih_besar) { kembalikan "A_MENDOMINASI"; }
    jika ada_b_lebih_besar dan (bukan ada_a_lebih_besar) { kembalikan "B_MENDOMINASI"; }
    kembalikan "KONFLIK_SEJAJAR"; // Butuh penggabungan (merge) manual
}

11.1 Arsitektur Internal B-Tree & Pembelahan Halaman (Page Split)

Ketika jumlah baris data dalam satu halaman B-Tree melebihi kapasitas 4096 bita, WidyaBase melakukan pembelahan halaman (page split) secara atomik. Kunci median dinaikkan ke halaman induk (parent page) untuk mempertahankan kedalaman pohon tetap seimbang (balanced tree depth $O(\log N)$):

11.2 Kueri Manufaktur Rantai Pasok: Bill of Materials (BOM)

impor "./modul/widyabase.wya";

fungsi analisis_kebutuhan_komponen_bom() {
    misal db = BukaWidyaBase("pabrik_otomotif.wdb");

    // CTE Rekursif untuk Menghitung Seluruh Sub-Komponen Kendaraan Listrik
    misal sql_bom = "
        DENGAN REKURSIF rincian_bom SEBAGAI (
            PILIH id_part, nama_komponen, jumlah, 1 SEBAGAI tingkat_sub
            DARI komponen_rakitan
            DIMANA id_induk ADALAH NIHIL

            GABUNG_SEMUA

            PILIH k.id_part, k.nama_komponen, r.jumlah * k.jumlah, r.tingkat_sub + 1
            DARI komponen_rakitan AS k
            GABUNG rincian_bom AS r PADA k.id_induk = r.id_part
        )
        PILIH * DARI rincian_bom URUTKAN_BERDASARKAN tingkat_sub NAIK;
    ";

    misal daftar_komponen = db.kueri(sql_bom);
    untuk p dalam daftar_komponen {
        cetak("Tingkat:", p["tingkat_sub"], "| Part:", p["nama_komponen"], "| Total Qty:", p["jumlah"]);
    }
}

11.3 Kueri Geospasial: Pencarian Armada Taksi Terdekat

fungsi cari_armada_terdekat(lintang_penumpang, bujur_penumpang) {
    misal db = BukaWidyaBase("transportasi_online.wdb");

    // Menghitung jarak geodesik menggunakan fungsi geometris bawaan WidyaBase
    misal sql_jarak = "
        PILIH 
            id_driver, 
            nama_supir, 
            ST_Jarak(lokasi_gps, ST_Titik(" + ke_teks(lintang_penumpang) + ", " + ke_teks(bujur_penumpang) + ")) AS radius_meter
        DARI armada_taksi
        DIMANA status_siaga = BENAR
        URUTKAN_BERDASARKAN radius_meter NAIK
        BATAS 3;
    ";

    misal supir_terpilih = db.kueri(sql_jarak);
    cetak("=== 3 ARMADA TERDEKAT DENGAN ANDA ===");
    untuk s dalam supir_terpilih {
        cetak("Supir:", s["nama_supir"], "| Jarak:", s["radius_meter"], "meter");
    }
}
